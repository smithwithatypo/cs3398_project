5. Connect React to Flask
🔹 Fetch data from Flask in React:
29. Open RecipeCard.js.
30. Replace the code with this:

jsx
Copy
Edit
import { useEffect, useState } from "react";

function RecipeCard() {
    const [message, setMessage] = useState("");

    useEffect(() => {
        fetch("http://127.0.0.1:5000/")
            .then(response => response.json())
            .then(data => setMessage(data.message));
    }, []);

    return (
        <div>
            <h2>Recipe Name</h2>
            <p>Ingredients: List of ingredients here</p>
            <p>Message from Backend: {message}</p>
        </div>
    );
}
export default RecipeCard;
Save the file and restart the React server (npm start).
Now your Flask backend is talking to your React frontend! 🎉


6. Create Recipe API in Flask
🔹 Modify app.py to return recipes:
33. Replace app.py with this:

python
Copy
Edit
from flask import Flask, jsonify

app = Flask(__name__)

recipes = [
    {"id": 1, "name": "Spaghetti", "ingredients": ["Pasta", "Tomato Sauce", "Garlic"]},
    {"id": 2, "name": "Omelette", "ingredients": ["Eggs", "Salt", "Pepper"]},
]

@app.route("/api/recipes")
def get_recipes():
    return jsonify(recipes)

if __name__ == "__main__":
    app.run(debug=True)
Restart Flask (python app.py).
Open a browser and visit http://127.0.0.1:5000/api/recipes. You should see the recipe list.


7. Display Recipes from Flask in React
🔹 Modify RecipeCard.js to display recipes from Flask:
36. Replace RecipeCard.js with:

jsx
Copy
Edit
import { useEffect, useState } from "react";

function RecipeCard() {
    const [recipes, setRecipes] = useState([]);

    useEffect(() => {
        fetch("http://127.0.0.1:5000/api/recipes")
            .then(response => response.json())
            .then(data => setRecipes(data));
    }, []);

    return (
        <div>
            <h1>Recipes</h1>
            {recipes.map(recipe => (
                <div key={recipe.id}>
                    <h2>{recipe.name}</h2>
                    <p>Ingredients: {recipe.ingredients.join(", ")}</p>
                </div>
            ))}
        </div>
    );
}
export default RecipeCard;
Save and restart React (npm start).
Now recipes from Flask appear on your webpage! 🎉
